import java.util.Arrays;

class QuickSort{
    public static int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;

        for (int j = low; j < high; j++) {
            if (arr[j] <= pivot) {
                i += 1;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }

        // Fixed typo: changed temp2 to temp1
        int temp1 = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp1;

        return i + 1;
    }

    // Overloaded helper method so you can call quicksort(arr) without low/high
    public static void quicksort(int[] arr) {
        quicksort(arr, 0, arr.length - 1);
    }

    // Standard Java parameters (no default values or semicolons)
    public static void quicksort(int[] arr, int low, int high) {
        if (low < high) {
            int pivot_index = partition(arr, low, high);
            quicksort(arr, low, pivot_index - 1);
            quicksort(arr, pivot_index + 1, high); // Fixed typo: quiclsort -> quicksort
        }
    }

    public static void main(String[] args) {
        int[] arr = {38, 27, 43, 3, 9, 82, 10};
        quicksort(arr); // Sorts arr in place
        System.out.println("Sorted Array: " + Arrays.toString(arr));
    }
}
